```bash

IMAGE_TAG=20261027-mall-release docker compose -f docker-compose.prod.yml \
  --project-name wuling up -d --no-build gateway auth-service file-service marketing-service user-service

  --project-name wuling up -d --no-build product-service

  --project-name wuling up -d --no-build trade-service

  --project-name wuling up -d --no-build server
```

```bash
mv /opt/wuling/web /opt/wuling/web.failed-20261101-all-services

```
